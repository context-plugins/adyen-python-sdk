| Name | Type | Tags | Description |

from adyen.models.fraud_check_result import FraudCheckResult

fraud_result_2 = FraudResult2(
    account_score=186,
    results=[
        FraudCheckResultWrapper(
            fraud_check_result=FraudCheckResult(
                account_score=114,
                check_id=2,
                name='name0'
            )
        ),
        FraudCheckResultWrapper(
            fraud_check_result=FraudCheckResult(
                account_score=114,

            )
        )

)
